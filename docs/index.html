API Reference & Code Guides

ตัวอย่างการเรียกใช้งานฟังก์ชันหลักของ Chronera ในโปรเจกต์ของคุณ 

import { localDate, formatDate, convertCalendarDate } from "@intech-software/chronera";

// 1. Create a clean date primitive
const date = localDate(2026, 9, 4);

// 2. Format with standard presets
console.log(formatDate(date, { dateStyle: "long" })); 
// -> "September 4, 2026"

// 3. Convert to Thai Buddhist calendar
const thai = convertCalendarDate(date, "buddhist");
console.log(thai.value.year); 
// -> 2569 (พ.ศ.)
import { isBefore, isEqual, isBetween, addDays, endOfMonth } from "@intech-software/chronera";

const d1 = localDate(2026, 9, 4);
const d2 = localDate(2026, 9, 10);

// Comparisons
isBefore(d1, d2); // true
isEqual(d1, d2);  // false

// Range checking
isBetween(d1, localDate(2026, 9, 1), localDate(2026, 9, 30)); // true

// Arithmetic shortcuts
const nextWeek = addDays(d1, 7); // 2026-09-11
const monthEnd = endOfMonth(d1);  // 2026-09-30
import { startOfDay, endOfDay, isPast, isFuture, addHours } from "@intech-software/chronera";

const today = localDate(2026, 9, 4);

// Time boundaries
startOfDay(today); // 2026-09-04 00:00:00.000
endOfDay(today);   // 2026-09-04 23:59:59.999

// Expiration checking (e.g. Tokens, Coupons)
const expireAt = new Date("2026-09-01T00:00:00Z");
isPast(expireAt); // true (Coupon expired!)
import { isWeekend, isWeekday, addBusinessDays, diffInBusinessDays } from "@intech-software/chronera";

const friday = localDate(2026, 9, 4); // Friday

// Adding 2 business days skips Saturday and Sunday!
const delivery = addBusinessDays(friday, 2); 
// -> 2026-09-08 (Tuesday)

// Signed business days difference
const diff = diffInBusinessDays(delivery, friday);
// -> 2 business days
import { formatInTimeZone, getTimeZoneOffset, isSameTimeZone } from "@intech-software/chronera";

const instant = new Date("2026-09-04T12:00:00Z");

// Format instant directly into target time zone
formatInTimeZone(instant, "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss XXX");
// -> "2026-09-04 19:00:00 +07:00"

formatInTimeZone(instant, "America/New_York", "yyyy-MM-dd HH:mm:ss (zzzz)");
// -> "2026-09-04 08:00:00 (Eastern Daylight Time)"

// Retrieve offset details (including DST)
getTimeZoneOffset("America/New_York", instant, "object");
// -> { formatted: "-04:00", minutes: -240, totalSeconds: -14400, isDst: true }

// Compare zones
isSameTimeZone("UTC", "Etc/UTC"); // true (Canonical match)
isSameTimeZone("Asia/Bangkok", "Asia/Jakarta", "offset"); // true (Both UTC+7)
